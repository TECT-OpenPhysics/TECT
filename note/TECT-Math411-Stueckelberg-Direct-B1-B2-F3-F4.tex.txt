\section{Setup: Stueckelberg identification from Math408}
==========================================================================================

\paragraph{The Stueckelberg field.} Math408 §3 proposed identifying the gauge field $A_\mu^a$ with the gradient of the $T_{2g}$ triplet fluctuation:
\[
A_\mu^a := \frac{1}{m_*} \partial_\mu \delta\phi^a, \quad a = 1, 2, 3.
\]
The Stueckelberg auxiliary $\chi^a = \delta\phi^a / m_*$ acts as the longitudinal scalar mode that restores gauge invariance:
\[
A_\mu^a \to A_\mu^a + \partial_\mu \alpha^a, \quad \chi^a \to \chi^a + \alpha^a / m_*.
\]
This is the standard Stueckelberg construction for a massive vector field.

\paragraph{Why Stueckelberg-direct AFTER Pathway B refutation.} Math410-AddA established that the SO(4) emergence via cubic-irrelevance is structurally blocked (A^* > 0 at exact-RG fixed point). The "doubled" SU(2)_L × SU(2)_R adjoint identification thus fails. The Stueckelberg-direct route is the simpler alternative: identify $T_{2g}$ as a SINGLE SU(2) gauge field, without doubling.

\paragraph{Required Yang-Mills structure}. For $A_\mu^a$ to be identified as the gauge field of an SU(2) Yang-Mills theory, the effective Lagrangian must contain:
\begin{align}
\mathcal{L}_{\rm YM} = & -\frac{1}{4} F_{\mu\nu}^a F^{a\mu\nu} \\
& + \frac{m_*^2}{2} (A_\mu^a - \partial_\mu \chi^a)^2 \\
& + f^{abc} A_\mu^b A_\nu^c \partial^\mu A^{a\nu},
\end{align}
where $F_{\mu\nu}^a = \partial_\mu A_\nu^a - \partial_\nu A_\mu^a + g f^{abc} A_\mu^b A_\nu^c$ and $f^{abc} = \epsilon^{abc}$ for SU(2).

\section{Gate B1: Maxwell kinetic-term coefficient}
==========================================================================================

\paragraph{Starting from BCC kinetic Lagrangian.} The Brazovskii kinetic structure at the $T_{2g}$ triplet is
\[
\mathcal{L}_{\rm kin}^{(T_{2g})} = \frac{1}{2}(\partial_\mu \delta\phi^a)^2 + \frac{1}{2} m_*^2 (\delta\phi^a)^2 + \frac{1}{2} \gamma (\Delta + q_*^2)^2 (\delta\phi^a)^2.
\]
The third term is the Brazovskii dispersion $(\Delta + q_*^2)^2$ which suppresses modes away from the shell $|q| = q_*$.

\paragraph{Substituting $A_\mu^a := \partial_\mu \delta\phi^a / m_*$.}
\[
\delta\phi^a = m_* \int A_\mu^a dx^\mu \quad \text{(in Lorenz gauge)}.
\]
The kinetic term becomes
\[
\frac{1}{2}(\partial_\mu \delta\phi^a)^2 = \frac{1}{2} m_*^2 (\partial_\mu \chi^a)^2 = \frac{1}{2} m_*^2 (\chi^a_{,\mu})^2.
\]
This is a SCALAR kinetic term for the Stueckelberg auxiliary, not the Maxwell form $-\frac{1}{4} F_{\mu\nu}^a F^{a\mu\nu}$.

\paragraph{Effective Lagrangian for $A_\mu^a$.} Working backward from $A_\mu^a$ definition:
\[
\partial_\mu A_\nu^a - \partial_\nu A_\mu^a = \frac{1}{m_*}(\partial_\mu \partial_\nu \delta\phi^a - \partial_\nu \partial_\mu \delta\phi^a) = 0 \quad \text{(partial derivatives commute)}.
\]
So $F_{\mu\nu}^a \equiv 0$ for the pure-Stueckelberg identification! The abelian field strength VANISHES, which means there is NO Maxwell-form kinetic term at leading order.

\paragraph{Honest interpretation.} The Stueckelberg-direct identification gives $A_\mu^a$ as a pure-gradient field (gauge $A_\mu^a = \partial_\mu \alpha^a$), which is in fact GAUGE-EQUIVALENT TO ZERO under the Stueckelberg gauge transformation. This is the well-known feature of the Stueckelberg construction: for an abelian massive vector, the gauge-fixed Lagrangian reproduces Proca-Stueckelberg structure, but the FIELD STRENGTH ITSELF vanishes for the pure-gradient identification.

\paragraph{Maxwell form emerges only at non-linear order.} For non-abelian extension, $F_{\mu\nu}^a$ contains the $f^{abc} A^b A^c$ commutator term. If $f^{abc} \neq 0$, then $F_{\mu\nu}^a \neq 0$ even for the Stueckelberg identification (since $\partial_\mu A_\nu^a - \partial_\nu A_\mu^a + g f^{abc} A_\mu^b A_\nu^c = 0$ has only trivial solution if $f^{abc} = 0$, but with $f^{abc} = \epsilon^{abc}$ gives non-trivial $F_{\mu\nu}^a$).
