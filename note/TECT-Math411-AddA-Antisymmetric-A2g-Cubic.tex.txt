\section{Pure-cubic antisymmetric identity}
==========================================================================================

\paragraph{Algebraic vanishing.} The natural candidate for a pure-cubic $A_{2g}$ operator is
\[
\mathcal{O}_{\rm pure}^{A_{2g}} = \epsilon^{abc} \phi^a \phi^b \phi^c.
\]
For commuting (bosonic) scalar field components $\phi^a$, this vanishes identically by index symmetry:
\[
\epsilon^{abc} \phi^a \phi^b \phi^c = \epsilon^{abc} (\phi^a \phi^b) \phi^c = -\epsilon^{bac} (\phi^a \phi^b) \phi^c = -\epsilon^{abc} \phi^b \phi^a \phi^c = -\epsilon^{abc} \phi^a \phi^b \phi^c,
\]
where the last step uses $\phi^a \phi^b = \phi^b \phi^a$ (bosonic commutativity). Therefore $2 \cdot \mathcal{O}_{\rm pure}^{A_{2g}} = 0$, i.e., $\mathcal{O}_{\rm pure}^{A_{2g}} \equiv 0$.

\paragraph{Implication.} \emph{No pure-cubic algebraic operator can carry the antisymmetric $A_{2g}$ singlet structure}. The $A_{2g}$ projection in $T_{2g}^{\otimes 3}$ requires derivatives — the structure constants of SU(2) Yang-Mills are intrinsically derivative interactions, not pure polynomial in $\phi$.
