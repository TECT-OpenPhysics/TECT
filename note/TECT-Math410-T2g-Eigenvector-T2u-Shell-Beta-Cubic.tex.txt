\section{T_{2g} eigenvector basis: explicit construction}
==========================================================================================

\paragraph{Setup.} Label the 6 amplitude modes as $\xi_a$ ($a = 1, \ldots, 6$) where each mode corresponds to a real linear combination of the antipodal pair's Fourier amplitudes:
\begin{align}
\xi_1 &= \mathrm{Re}\,\hat\phi(\vec{q}_1), & \xi_2 &= \mathrm{Im}\,\hat\phi(\vec{q}_1), \\
\xi_3 &= \mathrm{Re}\,\hat\phi(\vec{q}_2), & \xi_4 &= \mathrm{Im}\,\hat\phi(\vec{q}_2), \\
\xi_5 &= \mathrm{Re}\,\hat\phi(\vec{q}_3), & \xi_6 &= \mathrm{Im}\,\hat\phi(\vec{q}_3).
\end{align}
Under $O_h$, the symmetry generators act on the indices: $C_3$ rotations cycle $(\vec{q}_1, \vec{q}_2, \vec{q}_3)$; $C_4$ rotations swap pairs and apply phase; inversion $i$ flips all $\vec{q}_i \to -\vec{q}_i$ (which complex-conjugates the Fourier amplitudes, swapping $\xi_{2k-1} \leftrightarrow \xi_{2k-1}$ and negating $\xi_{2k} \to -\xi_{2k}$).

\paragraph{T_{2g} basis construction.} The $T_{2g}$ irrep transforms as the symmetric traceless tensor components $\{xy, yz, zx\}$. The corresponding combinations of the BCC amplitudes are:
\begin{align}
T_{xy} &= \frac{1}{\sqrt{2}}(\xi_1 - \xi_5), \\
T_{yz} &= \frac{1}{\sqrt{2}}(\xi_3 - \xi_1), \\
T_{zx} &= \frac{1}{\sqrt{2}}(\xi_5 - \xi_3).
\end{align}
\textbf{Constraint check}: $T_{xy} + T_{yz} + T_{zx} = 0$ (traceless), confirming this is a 3-dim subspace orthogonal to the $A_{1g}$ direction $A = (\xi_1 + \xi_3 + \xi_5)/\sqrt{3}$.

\paragraph{Verification of T_{2g} transformation.} Under $C_3$ rotation about the body-diagonal $(1,1,1)$, the index permutation is $(1 \to 3 \to 5 \to 1)$, which sends
\[
T_{xy} \to T_{yz} \to T_{zx} \to T_{xy}.
\]
This is the correct $T_{2g}$ cyclic permutation. Under $C_4$ about the $z$-axis, $T_{xy} \to -T_{xy}$, $T_{yz} \to T_{zx}$, $T_{zx} \to -T_{yz}$ (sign-flips matching the $T_{2g}$ character). \checkmark

\paragraph{Imaginary-part directions and the E_g doublet.}
The $E_g$ doublet, transforming as $\{x^2 - y^2, 2z^2 - x^2 - y^2\}/\sqrt{3}$, is constructed from the imaginary-part combinations:
\begin{align}
E_1 &= \frac{1}{\sqrt{2}}(\xi_2 - \xi_4), \\
E_2 &= \frac{1}{\sqrt{6}}(\xi_2 + \xi_4 - 2\xi_6).
\end{align}
The remaining direction is the $A_{1g}$ breathing mode:
\[
A_{1g} = \frac{1}{\sqrt{3}}(\xi_1 + \xi_3 + \xi_5) \text{ (real-part symmetric)}.
\]
The orthogonal $A_{1g}$ imaginary-part combination $(\xi_2 + \xi_4 + \xi_6)/\sqrt{3}$ is also present but corresponds to a global phase mode that decouples from physical observables.

\paragraph{Verification of dimension count}:
\[
\dim(A_{1g}) + \dim(E_g) + \dim(T_{2g}) = 1 + 2 + 3 = 6. \;\checkmark
\]

\paragraph{Setup.} The cubic anisotropy operator in the Brazovskii effective Lagrangian is
\[
\mathcal{O}_{\rm cubic} = A \sum_{i=1}^{6} (\vec{q}_i \cdot \vec{q}_i)^2 |\hat\phi(\vec{q}_i)|^2 \equiv A \, q_*^4 \, \xi^2_{\rm cubic},
\]
with $A$ the cubic coupling whose RG flow determines IR isotropy. At tree level, $A = A_0$ is set by the bare TECT Lagrangian.

\paragraph{One-loop correction.} Brazovskii shell-mode fluctuations renormalise $A$ via the bubble diagram:
\[
\delta A^{\rm 1-loop} = -\frac{1}{2} \lambda_4 \int \frac{d^3 k}{(2\pi)^3} \frac{1}{\left[(k^2 - q_*^2)^2 + r_R/\gamma\right]^2} \cdot (\vec{k} \cdot \vec{q}_*)^2.
\]
The angular integration projects the $(\vec{k} \cdot \vec{q}_*)^2$ onto the cubic anisotropy direction. The radial integral is dominated by the shell at $|k| = q_*$, with shell-thickness $\sim r_R^{1/4}/q_*^{1/2}$.

\paragraph{Shell-mode evaluation.} Expanding around $|k| = q_* + u$:
\begin{align}
\delta A^{\rm 1-loop} &= -\frac{\lambda_4}{2} \cdot \frac{4\pi q_*^2}{(2\pi)^3} \cdot \int du \, \frac{q_*^2 \langle \cos^2\theta \rangle}{[4 q_*^2 u^2 + r_R/\gamma]^2} \\
&= -\frac{\lambda_4 q_*^4}{(2\pi)^2} \cdot \langle \cos^2\theta \rangle \cdot \frac{\pi}{2 (r_R/\gamma)^{3/2} \cdot (4 q_*^2)} \\
&= -\frac{\lambda_4 q_*^2 \langle \cos^2\theta \rangle}{8\pi (r_R/\gamma)^{3/2}}.
\end{align}
With the cubic-projection $\langle \cos^2\theta \rangle_{\rm cubic} = 1/3$ for isotropic averaging (the cubic anisotropy direction picks out 1/3 of the spherical average):
\[
\delta A^{\rm 1-loop} = -\frac{\lambda_4 q_*^2}{24 \pi (r_R/\gamma)^{3/2}}.
\]

\paragraph{TECT canonical $\lambda_4 < 0$ (Reading H).} Substituting $\lambda_4 = -|\lambda_4|$ (attractive quartic in Reading H regime, Math401):
\[
\boxed{
\delta A^{\rm 1-loop} = +\frac{|\lambda_4| \, q_*^2}{24 \pi (r_R/\gamma)^{3/2}}.

\]

\[
\beta_A = k \frac{\partial A}{\partial k} = -A + \delta A^{\rm 1-loop} \cdot k^2/q_*^2 \quad (\text{schematic Wilson form}).
\]
For positive $\delta A^{\rm 1-loop}$, the cubic coupling RUNS TOWARD A POSITIVE FIXED POINT at $k = 0$, not to zero. This means $O_h$ symmetry is RETAINED in the IR (negative result for Pathway B isotropy emergence).
