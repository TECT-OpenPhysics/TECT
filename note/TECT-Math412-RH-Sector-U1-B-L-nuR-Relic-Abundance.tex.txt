\section{ν_R identification in the 16-rep}
==========================================================================================

\paragraph{$\nu_R$ as singlet.} The 16-rep contains a SM-singlet component (no color, no SU(2)$_L$, no hypercharge) carrying $(B-L) = +1$. This is the right-handed neutrino $\nu_R$:
\[
\nu_R = (\mathbf{1}, \mathbf{1}, 0; +1)_{\rm SM \times U(1)_{B-L}}.
\]
With three generations (matching three SM lepton flavours), TECT predicts $N_{\nu_R} = 3$.

\paragraph{Majorana mass term from $v_R$.} Under SU(2)$_R$ breaking by $v_R$, the $\nu_R$ acquires a Majorana mass via the Yukawa coupling to the SU(2)$_R$ triplet Higgs $\Delta_R$:
\[
\mathcal{L}_{\nu_R-{\rm mass}} = y_{\nu_R} \,\overline{\nu_R^c}\,\Delta_R\,\nu_R \;\to\; \frac{M_{\nu_R}}{2}\,\overline{\nu_R^c}\,\nu_R, \quad M_{\nu_R} = y_{\nu_R}\,v_R.
\]
For $y_{\nu_R} \sim O(1)$ (natural Yukawa) and $v_R \sim 10^{14}$ GeV:
\[
\boxed{M_{\nu_R} \approx 10^{14}\,{\rm GeV} = 10^{-5}\,M_{\rm Pl}.}
\]

\section{ν_R relic abundance via gravitational reheating}
==========================================================================================

\paragraph{Production mechanism.} Post-inflation reheating creates a thermal bath at $T_{\rm reh}$. For TECT $\nu_R$ that is decoupled from SM gauge bosons (it's a SM singlet), the only production channel is gravitational: $\nu_R$ pairs created from inflaton decay via graviton exchange.

\paragraph{Asaka-Shaposhnikov-class formula.} The yield from gravitational reheating (Mishra, Modak, Mohapatra 2018; Garcia, Ko, Park 2019):
\[
Y_{\nu_R} \equiv \frac{n_{\nu_R}}{s} \approx \frac{N_{\nu_R}}{g_*(T_{\rm reh})} \cdot \left(\frac{T_{\rm reh}}{M_{\rm Pl}}\right)^3 \cdot \alpha_{\rm grav},
\]
where $g_*(T_{\rm reh}) \approx 200$ at GUT scale (SM + extended sector), $N_{\nu_R} = 3$ generations, and $\alpha_{\rm grav} \sim 10^{-3}$ is the gravitational production efficiency (model-dependent).

\paragraph{Relic abundance.} Today's $\nu_R$ density:
\[
\Omega_{\nu_R} h^2 = M_{\nu_R} \cdot Y_{\nu_R} \cdot \frac{s_0}{\rho_{c}/h^2} = 2.74 \times 10^8 \cdot M_{\nu_R}\,({\rm GeV}) \cdot Y_{\nu_R}.
\]

\paragraph{Required reheating temperature for $\Omega_{\nu_R} h^2 = 0.12$.} Setting $\Omega_{\nu_R} = \Omega_{\rm DM}$ and solving for $T_{\rm reh}$:
\begin{align}
Y_{\nu_R}^{\rm required} &= \frac{0.12}{M_{\nu_R}\,({\rm GeV}) \cdot 2.74 \times 10^8} = \frac{0.12}{10^{14} \cdot 2.74 \times 10^8} = 4.4 \times 10^{-24}, \\
\frac{T_{\rm reh}^3}{M_{\rm Pl}^3} &\approx \frac{g_*\,Y_{\nu_R}^{\rm required}}{N_{\nu_R} \cdot \alpha_{\rm grav}} = \frac{200 \cdot 4.4 \times 10^{-24}}{3 \cdot 10^{-3}} = 2.9 \times 10^{-19}, \\
T_{\rm reh} &\approx (2.9 \times 10^{-19})^{1/3} \cdot M_{\rm Pl} = 6.6 \times 10^{-7} \cdot M_{\rm Pl} \approx 8 \times 10^{12}\,{\rm GeV}.
\end{align}

\paragraph{Closure window} (with factor 3 uncertainty in $\alpha_{\rm grav}$):
\[
\boxed{
T_{\rm reh}^{\rm required} \;\in\; [3 \times 10^{12},\; 2 \times 10^{13}]\,{\rm GeV}.
}
\]

\paragraph{Objection β (gravitational vs Yukawa channels)}. Gravitational $\nu_R$ production has efficiency $\alpha_{\rm grav} \sim 10^{-3}$ (model-dependent, factor 10 uncertainty). Yukawa production (if any) would dominate.

PASS at the chosen parameter range.

\paragraph{Objection γ (warm-DM constraints)}. Heavy $\nu_R$ ($M \sim 10^{14}$ GeV) is COLD DM (large mass, low velocity). However, if $\nu_R$ decays into lighter SM particles via small mixing angle, the daughter SM neutrinos could be warm/hot, affecting structure formation.

Seesaw mixing implies $\nu_R$ is NOT stable on cosmological timescales. \textbf{This is a serious problem}: for $\nu_R$ to be DM, it must be stable. Either $M_{\nu_R}$ must be much smaller (keV scale, sterile neutrino DM) so that decay is kinematically suppressed, or the active-sterile mixing must be zero (no seesaw).
