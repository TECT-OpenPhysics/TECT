\section{ν_R decay channels and rate formula}
==========================================================================================

\paragraph{Active-sterile mixing.} The sterile $\nu_R$ mixes with the active $\nu_L$ via a small angle $\theta$. The mass eigenstates are
\[
\nu_1 = \cos\theta\,\nu_L - \sin\theta\,\nu_R, \quad \nu_2 = \sin\theta\,\nu_L + \cos\theta\,\nu_R.
\]
For seesaw type I: $\sin\theta \approx m_D/M_R = y v/M_R$ where $m_D$ is the Dirac mass and $M_R$ the Majorana mass. The light eigenstate has $m_1 \approx -m_D^2/M_R = -y^2 v^2/M_R$.

\paragraph{Observed light-neutrino mass.} Setting $|m_1| = m_{\rm active} \approx 0.05$ eV $= 5 \times 10^{-11}$ GeV (from atmospheric oscillation):
\[
y^2 = \frac{m_{\rm active} M_R}{v^2}.
\]

\paragraph{Decay rate (3-body dominant)}:
\[
\Gamma(\nu_R \to 3\nu) = \frac{G_F^2 M_{\nu_R}^5 \sin^2(2\theta)}{192\pi^3}.
\]

\paragraph{Lifetime}:
\[
\tau_{\nu_R} = \frac{1}{\Gamma} = \frac{192\pi^3}{G_F^2 M_{\nu_R}^5 \sin^2(2\theta)}.
\]
Using $G_F = 1.166 \times 10^{-5}$ GeV$^{-2}$ and $\hbar = 6.58 \times 10^{-25}$ GeV s:
\[
\tau_{\nu_R}\,({\rm s}) \approx \frac{1.1 \times 10^{20}}{(M_{\nu_R}\,({\rm GeV}))^5 \cdot \sin^2(2\theta)}.
\]
