\section{Path A: BCC-channel coupling derivation}
==========================================================================================

\paragraph{LRSM cascade structure (Math408 §5)}. The Standard-Model gauge group $G_{\rm SM} = SU(3)_C \times SU(2)_L \times U(1)_Y$ emerges from SO(10) via the chain
\[
SO(10) \;\supset\; SU(5) \times U(1)_\chi \;\supset\; G_{\rm SM} \times U(1)_{B-L},
\]
where $U(1)_{B-L}$ is broken at $v_R \sim 10^{14}$ GeV by the right-handed Higgs $\Delta_R$ (an LRSM triplet). The breaking gives Majorana mass $M_{\nu_R} \sim v_R$ to the right-handed neutrino via $\Delta_R \nu_R^c \nu_R$.

\paragraph{$\Delta_R$ as BCC-channel-coupled field}. In TECT's Reading H framework (Math401 binding), the LRSM Higgs $\Delta_R$ is identified with the BCC channel content at scale $v_R$ via the cascade (Math408 §5 + Math412 §3). Specifically, $\Delta_R$ acquires its VEV from the BCC channel fluctuation amplitude $\langle |\delta\Psi(q_*)|^2 \rangle$ at the LRSM scale.

\paragraph{$\mathbb{Z}_2$ charge inheritance}. The Krauss-Wilczek $\mathbb{Z}_2$ acts on fields that wind around the $\pi_1 = \mathbb{Z}_2$ cosmic strings. Fields coupled to the BCC channel (and therefore to the cosmic strings supported by $\pi_1(V) = \mathbb{Z}_2$) inherit the $\mathbb{Z}_2$ charge. Specifically:
\[
\Delta_R \to -\Delta_R \quad\text{under}\quad \mathbb{Z}_2 \quad (\text{charge } +1).
\]
And via the Yukawa coupling $\bar{\nu}_R^c \Delta_R \nu_R$:
\[
\nu_R \to -\nu_R \quad\text{under}\quad \mathbb{Z}_2 \quad (\text{charge } +1; \text{half-integer winding}).
\]
The active light neutrino $\nu_L$ is part of the SM $SU(2)_L$ doublet $L = (\nu_L, e_L)^T$; the SM Higgs $H$ breaks $SU(2)_L \times U(1)_Y \to U(1)_{\rm em}$ at $v_{\rm EW} = 246$ GeV (a SEPARATE scale, far below $v_R$). The SM Higgs $H$ is in the BCC-channel-NEUTRAL sector at the LRSM scale; therefore:
\[
H \to +H,\quad \nu_L \to +\nu_L \quad\text{under}\quad \mathbb{Z}_2 \quad (\text{charge } 0; \text{integer winding}).
\]

\paragraph{Resulting selection rule}. The Yukawa coupling $y \bar{L} H \nu_R$:
\[
y\bar{L} H \nu_R \to y (+\bar{L})(+H)(-\nu_R) = -y\bar{L} H \nu_R.
\]
This term is $\mathbb{Z}_2$-ODD. For the action to be $\mathbb{Z}_2$-invariant, either:
\begin{itemize}
\item[(i)] The term is forbidden $\Rightarrow y = 0$ — contradicts observed neutrino oscillation.
\item[(ii)] Alternative Yukawa structure via a $\mathbb{Z}_2$-charged Higgs $H_R$: $y' \bar{L} H_R \nu_R$ with $H_R \to -H_R$. Then the term is $\mathbb{Z}_2$-EVEN.
\end{itemize}

\paragraph{Realisation via $\Delta_R$ as $H_R$}. The LRSM $\Delta_R$ is the natural candidate for $H_R$. It is $\mathbb{Z}_2$-charged (per BCC coupling above) and gives the Majorana mass $\Delta_R \nu_R^c \nu_R$ which IS $\mathbb{Z}_2$-invariant (two $\nu_R$ + one $\Delta_R$: $(-1)(-1)(-1) = -1$... wait that's odd).

\textbf{Refined check}: Majorana mass term $\bar{\nu_R^c} \Delta_R \nu_R$. Under $\mathbb{Z}_2$: $\nu_R^c \to -\nu_R^c$, $\Delta_R \to -\Delta_R$, $\nu_R \to -\nu_R$. Product: $(-1)(-1)(-1) = -1$. ODD.

For Majorana mass to be $\mathbb{Z}_2$-invariant, the product must be even. With $\nu_R$ charge $+1$ and $\Delta_R$ charge $+1$, three-field product is odd. Need to assign $\Delta_R$ charge such that $2 \cdot (\text{charge}(\nu_R)) + \text{charge}(\Delta_R) = 0\pmod 2$:
\[
2 \cdot 1 + \text{charge}(\Delta_R) = 0\pmod 2 \;\Rightarrow\; \text{charge}(\Delta_R) = 0\pmod 2.
\]
So $\Delta_R$ must be $\mathbb{Z}_2$-NEUTRAL (charge $0$), not charge $+1$.

\textbf{Reconsideration}: if $\Delta_R$ is $\mathbb{Z}_2$-neutral and $\nu_R$ is $\mathbb{Z}_2$-charged, then $\bar{\nu_R^c} \Delta_R \nu_R$ has $\mathbb{Z}_2$-parity $(+1)(-1)^2 = +1$ — INVARIANT (two $\nu_R$ insertions cancel).
And the Dirac Yukawa $\bar{L} H \nu_R$ has parity $(+1)(+1)(-1) = -1$ — ODD.

So the Dirac Yukawa is forbidden by $\mathbb{Z}_2$ — unless via an alternative $\mathbb{Z}_2$-charged Higgs $H_R$:
\[
y' \bar{L} H_R \nu_R \text{ with } H_R \to -H_R \quad \Rightarrow \quad (+1)(-1)(-1) = +1 \text{ INVARIANT}.
\]

\paragraph{Net Path A structure}:
\begin{itemize}
\item $\nu_R$: $\mathbb{Z}_2$ charge $+1$ (non-trivial).
\item $\nu_L$, SM fields, SM Higgs $H$: $\mathbb{Z}_2$ charge $0$ (trivial).
\item $\Delta_R$ (Majorana mass Higgs): $\mathbb{Z}_2$ charge $0$ (neutral; gives invariant Majorana term).
\item $H_R$ (additional $\mathbb{Z}_2$-charged Higgs for Dirac Yukawa): $\mathbb{Z}_2$ charge $+1$. \textbf{This is a NEW TECT structural requirement}.
\end{itemize}
