\section{Standard bundle-topology derivation}
==========================================================================================

\paragraph{Chern-Pontryagin relation for SO(N) bundles}. For a principal SO(N) bundle $P$ over a 4-manifold $M^4$ with associated vector bundle $E_{\mathbf{R}}$ (in representation $\mathbf{R}$), the second Chern class is determined by:
\[
c_2(E_{\mathbf{R}}) \;=\; c_2(\mathbf{R}) \cdot p_1(P),
\]
where $c_2(\mathbf{R})$ is the quadratic Casimir / index of the representation $\mathbf{R}$, and $p_1(P)$ is the first Pontryagin class of the principal bundle.

\paragraph{For SO(10) spinor $\mathbf{16}$}: standard representation-theoretic computation gives
\[
c_2(\mathbf{16}) = 2 \quad (\text{Slansky 1981, Table 7; SO(10) spinor index})
\]
in the natural normalisation where the fundamental $\mathbf{10}$-rep has $c_2(\mathbf{10}) = 1$.

\paragraph{Therefore}:
\[
c_2(E_{\mathbf{16}}) = 2 \cdot p_1(P_{SO(10)}).
\]
For $c_2(E_{\mathbf{16}}) = 0$, we require $p_1(P_{SO(10)}) = 0$ — i.e., the SO(10) principal bundle has trivial first Pontryagin number.
